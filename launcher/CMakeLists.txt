cmake_minimum_required(VERSION 3.20)
project(VoiceCoverStudio LANGUAGES C)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Build the desktop launcher for 64-bit Windows.")
endif()
enable_language(RC)
add_executable(VoiceCoverStudio WIN32 entry.c icon.rc)
target_include_directories(VoiceCoverStudio PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../voice_studio/assets")
target_compile_definitions(VoiceCoverStudio PRIVATE UNICODE _UNICODE)
target_link_libraries(VoiceCoverStudio PRIVATE user32 kernel32)
if(MSVC)
    target_compile_options(VoiceCoverStudio PRIVATE $<$<COMPILE_LANGUAGE:C>:/utf-8>)
    set_property(TARGET VoiceCoverStudio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
    target_link_options(VoiceCoverStudio PRIVATE -municode)
endif()
